\begin{frame}{研究问题}
\key{专家独立使用时保持能力，未经授权融合后降低被保护任务性能。}
\[
\theta_{\rm release}=T_c(\theta_e),\qquad
\theta_{\rm merge}=\theta_0+\tfrac12
\big[(\theta_{\rm release}-\theta_0)+(\theta_p-\theta_0)\big].
\]
\begin{itemize}\setlength{\itemsep}{6pt}
\item 已微调专家$\theta_e$冻结；保护发生在模型发布前。
\item 构造仅使用目标专家、公开基座及目标训练数据，不读取未来伙伴。
\item 同时评价\textbf{被保护任务与伙伴任务}，区分融合和攻击恢复。
\end{itemize}
\vspace{3mm}
两套方案：\textbf{投影核空间偏置扰动}；\textbf{任务子空间约束的跨模块补偿}。

\begin{frame}{方案一：投影核空间中的偏置扰动}
CLIP末端特征为$t=\operatorname{LN}_{\gamma,b}(h)P_e$，
$P_e\in\mathbb R^{768\times512}$。选择
\[
v^\top P_e=0,\qquad b'=b+\alpha v
\quad\Longrightarrow\quad t'=t.
\]
理想算术下独立特征不变；融合后读出矩阵改变，扰动变为
$\delta t_{\rm merge}=\frac{\alpha}{2}v^\top P_{\rm merge}$。

\begin{frame}{方案二：距离目标与任务内外的实际处理}
\small
任务向量$\Delta W=W_e-W_0=U\Sigma V^\top$；O/B旧参数化为
\[
\delta O=\|O_e\|_F U_{32}CV_{32}^\top,\qquad
\delta B=\|B_e\|_F Q_{\rm cls}K.
\]
\[
\max_c\;D(c)^2/0.5^2+\phi_{\rm source}(c)
\quad\mathrm{s.t.}\quad D(c)\le0.5,\quad R_{\mathcal C\cup\mathcal P}(c)\le0.5.
\]
\begin{itemize}\setlength{\itemsep}{5pt}
\item 距离最大化目标已经写入；$R$是完整分数的相对分类间隔变化，不是概率差。
\item 上游仅开放32维任务SVD坐标；B输入已覆盖3072维，输出仍受分类读出空间限制。
\item 奇异方向作有限上下文探针，另用自然训练输入验收；不能当作完整功能零空间。
\end{itemize}
\key{需要联合控制完整功能，不是简单删去任务内分量或放大噪声。}
\source{代码：angular\_distance.py、boundary\_game.py、ffn\_game.py。有限最坏任务约束及公开基座代理，不构成未知伙伴下的完整安全博弈；仅发布系数有梯度。}
\end{frame}
